% PON downstream broadcast and upstream TDMA (Section 2.1).
% Drawn by the author, adapted from the public-domain diagram by Roonie.02 (2013).
\definecolor{pkta}{RGB}{166,206,227}
\definecolor{pktb}{RGB}{178,223,138}
\definecolor{pktc}{RGB}{253,191,111}
\begin{tikzpicture}[
    font=\small,
    box/.style={draw, thick, rounded corners=2pt, minimum width=1.3cm,
                minimum height=0.75cm, fill=white},
    user/.style={draw, rounded corners=2pt, minimum width=1.1cm,
                 minimum height=0.6cm, fill=black!5, font=\footnotesize},
    split/.style={draw, thick, circle, minimum size=0.55cm, inner sep=0pt,
                  fill=white},
    fiber/.style={thick},
    pkt/.style={draw, rounded corners=1pt, minimum width=0.44cm,
                minimum height=0.44cm, inner sep=0pt, font=\scriptsize\bfseries},
    p1/.style={pkt, fill=pkta}, p2/.style={pkt, fill=pktb}, p3/.style={pkt, fill=pktc},
    flow/.style={-{Stealth[length=2.2mm]}, thick, black!70},
    lbl/.style={font=\footnotesize\itshape},
    title/.style={font=\small\bfseries, anchor=west}
  ]

  % ---------------- (a) downstream: broadcast ----------------
  \begin{scope}
    \node[title] at (-0.65,2.3) {(a) Downstream: broadcast};
    \node[box] (olt) at (0,0) {OLT};
    \node[split] (sp) at (4.0,0) {};
    \node[lbl, below=1pt of sp] {splitter};
    \draw[fiber] (olt) -- (sp);
    \foreach \k/\x in {3/1.5, 2/2.05, 1/2.6} { \node[p\k] at (\x,0) {\k}; }
    \draw[flow] (1.3,0.5) -- (2.9,0.5);
    \foreach \i/\y in {1/1.6, 2/0, 3/-1.6} {
      \node[box] (onu\i) at (9.4,\y) {ONU \i};
      \node[user] (usr\i) at (12.6,\y) {User \i};
      \draw[fiber] (sp) -- (onu\i.west);
      \draw[fiber] (onu\i.east) -- (usr\i.west);
      % every branch carries every frame ...
      \foreach \k/\t in {3/0.48, 2/0.63, 1/0.78} {
        \node[p\k] at ($(sp)!\t!(onu\i.west)$) {\k};
      }
      % ... and the ONU passes on only its own
      \node[p\i] at ($(onu\i.east)!0.5!(usr\i.west)$) {\i};
    }
    \node[lbl, anchor=north] at (9.4,-2.05)
      {every ONU receives every frame and keeps only its own};
  \end{scope}

  % ---------------- (b) upstream: TDMA ----------------
  \begin{scope}[yshift=-5.9cm]
    \node[title] at (-0.65,2.3) {(b) Upstream: TDMA};
    \node[box] (olt) at (0,0) {OLT};
    \node[split] (sp) at (4.0,0) {};
    \node[lbl, below=1pt of sp] {splitter};
    \draw[fiber] (olt) -- (sp);
    % the bursts reach the OLT one after the other, each in its own slot
    \foreach \k/\x in {3/1.3, 2/2.05, 1/2.8} { \node[p\k] at (\x,0) {\k}; }
    \draw[flow] (2.9,0.5) -- (1.3,0.5);
    \node[lbl, anchor=south] at (2.1,0.7) {one time slot per ONU (DBA)};
    \foreach \i/\y in {1/1.6, 2/0, 3/-1.6} {
      \node[box] (onu\i) at (9.4,\y) {ONU \i};
      \node[user] (usr\i) at (12.6,\y) {User \i};
      \draw[fiber] (sp) -- (onu\i.west);
      \draw[fiber] (onu\i.east) -- (usr\i.west);
      \node[p\i] at ($(sp)!0.6!(onu\i.west)$) {\i};
      \node[p\i] at ($(onu\i.east)!0.5!(usr\i.west)$) {\i};
    }
  \end{scope}
\end{tikzpicture}
