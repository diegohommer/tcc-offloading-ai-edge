% The prefill and decode phases of LLM inference (Section 2.2).
% Adapted from Zhou et al. (2024), Fig. 2; drawn by the author.
\begin{tikzpicture}[
    font=\small,
    tok/.style={draw, minimum width=0.62cm, minimum height=0.55cm,
                inner sep=0pt, font=\footnotesize},
    gen/.style={tok, fill=black!25},
    stage/.style={draw, thick, rounded corners=2pt, minimum height=1.0cm,
                  align=center, fill=white},
    cache/.style={draw, thick, fill=black!8, minimum height=0.6cm},
    flow/.style={-{Stealth[length=2mm]}, thick},
    note/.style={font=\footnotesize\itshape, align=center}
  ]

  % prompt tokens, processed together
  \foreach \k in {1,...,4} { \node[tok] (x\k) at (\k*0.66-0.66,0) {$x_{\k}$}; }
  \node[font=\footnotesize, above=2pt of x2.north east] {prompt};

  \node[stage, minimum width=2.0cm] (pre) at (4.1,0) {Prefill};
  \draw[flow] (x4.east) -- (pre.west);
  \node[gen] (y1) at (6.0,0) {$y_{1}$};
  \draw[flow] (pre.east) -- (y1.west);

  % decode steps, one token each
  \node[stage, minimum width=1.4cm] (d1) at (7.6,0) {Decode};
  \node[gen] (y2) at (9.2,0) {$y_{2}$};
  \node[stage, minimum width=1.4cm] (d2) at (10.8,0) {Decode};
  \node[gen] (y3) at (12.4,0) {$y_{3}$};
  \node at (13.25,0) {$\cdots$};
  \draw[flow] (y1.east) -- (d1.west);
  \draw[flow] (d1.east) -- (y2.west);
  \draw[flow] (y2.east) -- (d2.west);
  \draw[flow] (d2.east) -- (y3.west);

  % the KV cache under both phases
  \node[cache, minimum width=10.0cm, anchor=west] (kv) at (3.1,-1.9) {KV cache};
  \draw[flow] (pre.south) -- (pre.south |- kv.north)
    node[midway, right, font=\footnotesize] {store};
  \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]}, thick]
    (d1.south) -- (d1.south |- kv.north)
    node[midway, right, font=\footnotesize] {read, append};
  \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]}, thick]
    (d2.south) -- (d2.south |- kv.north)
    node[midway, right, font=\footnotesize] {read, append};

  % what bounds each phase
  \node[note] at (1.0,-1.2) {all prompt tokens\\at once: compute-bound};
  \draw[decorate, decoration={brace, amplitude=5pt, mirror}]
    (6.9,-2.45) -- (13.4,-2.45)
    node[midway, below=6pt, note] {one token per step: memory-bound};
\end{tikzpicture}
