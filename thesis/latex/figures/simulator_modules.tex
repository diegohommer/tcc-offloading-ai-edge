\begin{tikzpicture}[

    core/.style={module, fill=black!10},
    data/.style={draw, dashed, minimum width=2.8cm, minimum height=0.85cm, align=center, font=\footnotesize},

  % the core, with the two modules it asks
  \node[core] (loop) at (6.3,0) {Event loop};

  \draw[both] (loop) -- node[font=\footnotesize, align=left, right=2pt] {queries, answers,\\the OLT's report} (olt);
  \draw[flow] (energy) -- node[font=\footnotesize, align=center, below=1pt] {batch rates, speeds,\\$r_{\text{idle}}$, $r_{\text{busy}}$} (olt);
  \draw[flow] (energy) -- node[font=\footnotesize, sloped, above=1pt, pos=0.5] {rates and speeds} (loop);
  \draw[flow, dashed] (answers) -- (loop);
  \draw[flow, dashed] (loop) -- (results);
