% The conclusion's three findings as cards with a headline number (thesis
% Chapter "Conclusion" and Sections "Energy at the Same Accuracy", "Latency
% and Communication Burden"): the energy, red, then the wait and the traffic.
\begin{tikzpicture}[
    card/.style={draw=#1, line width=1pt, rounded corners=4pt, minimum width=4.4cm,
                 minimum height=3.5cm, fill=white, inner sep=0pt}
  ]
  % #1 x, #2 title (also the node's name), #3 number, #4 label, #5 color
  \newcommand{\findingcard}[5]{%
    \node[card=#5] (c#2) at (#1, 0) {};
    \fill[#5, rounded corners=4pt] ([yshift=-0.6cm]c#2.north west)
      -- (c#2.north west) -- (c#2.north east) -- ([yshift=-0.6cm]c#2.north east) -- cycle;
    \fill[#5] ([yshift=-0.6cm]c#2.north west) rectangle ([yshift=-0.3cm]c#2.north east);
    \node[text=white, font=\normalsize\bfseries] at ([yshift=-0.3cm]c#2.north) {#2};
    \node[font=\fontsize{22}{24}\selectfont\bfseries, text=#5] at ([yshift=-1.4cm]c#2.north) {#3};
    \node[text width=4.0cm, align=center, font=\small, text=InfGray, anchor=north]
      at ([yshift=-2.05cm]c#2.north) {#4};
  }
  \findingcard{0}{Energy}{$-$52\%}
    {against confidence alone,\\ same accuracy, 20,000 households}{InfRed}
  \findingcard{4.7}{Wait}{48 $\to$ 13 s}
