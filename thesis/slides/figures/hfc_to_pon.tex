% HFC and PON side by side: the cable network's CMTS, optical node and cable
% modem against the PON's OLT, passive splitter and ONU. Coax turns into fiber.
\begin{tikzpicture}[
    font=\footnotesize,
    endpoint/.style={draw, thick, rounded corners=2pt, minimum width=1.2cm,
                     minimum height=0.55cm, fill=white, font=\scriptsize},
    junction/.style={draw, thick, circle, minimum size=0.35cm, inner sep=0pt,
                     fill=white},
    fiber/.style={line width=1.2pt, InfRed},
    coax/.style={line width=3pt, InfLightGray},
    medium/.style={above=1pt, font=\tiny},
    under/.style={below=2pt, font=\tiny, text=InfGray},
    row/.style={anchor=east, font=\scriptsize\bfseries}
  ]
  \node[row] at (-0.8, 0.8) {HFC};
  \node[endpoint] (cmts) at (0, 0.8) {CMTS};
  \node[junction] (opticalnode) at (2.0, 0.8) {};
  \node[under] at (opticalnode.south) {node};
  \node[endpoint, minimum width=1.5cm] (modem) at (4.1, 0.8) {cable modem};
  \draw[fiber] (cmts) -- node[medium, text=InfRed] {fiber} (opticalnode);
  \draw[coax] (opticalnode) -- node[medium, text=InfGray] {coax} (modem);

  \node[row, text=InfRed] at (-0.8, -0.8) {PON};
  \node[endpoint] (olt) at (0, -0.8) {OLT};
  \node[junction] (splitter) at (2.0, -0.8) {};
  \node[under] at (splitter.south) {splitter};
  \node[endpoint, minimum width=1.5cm] (onu) at (4.1, -0.8) {ONU};
  \draw[fiber] (olt) -- node[medium, text=InfRed] {fiber} (splitter);
  \draw[fiber] (splitter) -- node[medium, text=InfRed] {fiber} (onu);
\end{tikzpicture}
