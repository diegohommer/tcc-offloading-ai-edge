% The OLT's GPU energy per token against the size of a static batch (thesis
% Table "Energy of the OLT's GPU in static batches", Joules per token): the
% decode (generated tokens) gets about 50 times cheaper from 1 to 64 queries,
% the prefill (prompt tokens) about 2 times. Plain TikZ, since pgfplots made
% the deck too slow for Overleaf: both axes are logarithmic, so a point is
% (log2 of the batch, log10 of the joules), one doubling per 0.66 cm and one
% decade per 1.2 cm.
\begin{tikzpicture}[x=0.66cm, y=1.2cm,
    tick/.style={font=\scriptsize, text=InfGray},
    line/.style={line width=1.4pt}]
  % Axes and ticks
  \draw[InfGray, -{Stealth[length=1.6mm]}] (-0.3,-2.15) -- (6.4,-2.15);
  \draw[InfGray, -{Stealth[length=1.6mm]}] (-0.3,-2.15) -- (-0.3,0.75);
  \foreach \batch/\at in {1/0,2/1,4/2,8/3,16/4,32/5,64/6} {
    \draw[InfGray] (\at,-2.15) -- (\at,-2.22);
    \node[tick, anchor=north] at (\at,-2.24) {\batch};
  }
  \foreach \joules/\at in {0.01/-2,0.1/-1,1/0} {
    \draw[InfGray] (-0.3,\at) -- (-0.39,\at);
    \node[tick, anchor=east] at (-0.42,\at) {\joules};
  }
  \node[font=\small, anchor=north] at (3,-2.55) {queries in the batch};
  \node[font=\small, rotate=90, anchor=south] at (-1.3,-0.7) {J per token, GPU card};
  % Generated tokens: the decode, limited by memory bandwidth
  \draw[line, InfRed] plot[mark=*, mark size=1.6pt] coordinates {
    (0.000,0.392) (1.000,0.089) (1.585,-0.084) (2.000,-0.209) (2.585,-0.382)
    (3.000,-0.504) (3.585,-0.676) (4.000,-0.796) (4.585,-0.959) (5.000,-1.076)
    (5.585,-1.222) (6.000,-1.328)};
  % Prompt tokens: the prefill, limited by computation
  \draw[line, InfGray] plot[mark=*, mark size=1.6pt] coordinates {
    (0.000,-1.481) (1.000,-1.409) (1.585,-1.538) (2.000,-1.620) (2.585,-1.721)
    (3.000,-1.770) (3.585,-1.796) (4.000,-1.824) (4.585,-1.824) (5.000,-1.824)
    (5.585,-1.854) (6.000,-1.854)};
  \node[anchor=south west, font=\small\bfseries, text=InfRed] at (0.15,0.43) {2.46~J};
  \node[anchor=west, font=\small\bfseries, text=InfRed] at (6.25,-1.328) {0.047~J};
  \node[anchor=south west, font=\small, text=InfRed] at (2.3,-0.3) {generated token};
  \node[anchor=south west, font=\small, text=InfGray] at (2.3,-1.66) {prompt token};
\end{tikzpicture}
